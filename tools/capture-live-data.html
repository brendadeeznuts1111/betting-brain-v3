// 🎯 BetTicker Live Capture Script
(async function() {
    console.log('🎯 Capturing betting data through interceptor...');
    
    try {
        // Make request to YOUR worker (not fantasy402 directly)
        const response = await fetch(
            'https://betting-brain-v3.nolarose1968-806.workers.dev/cloud/api/Manager/getBetTicker',
            {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/x-www-form-urlencoded',
                    // Auth headers will be sent automatically from fantasy402.com context
                },
                body: 'agent=&level=0&ticket=&customer=&daterange=01%2F01%2F1970+-+12%2F31%2F2026&show=&limit=200&offset=0',
                credentials: 'include'
            }
        );
        
        const data = await response.json();
        const wagerCount = data.LIST ? data.LIST.length : 0;
        
        console.log('✅ Success!');
        console.log('📊 Captured:', wagerCount, 'wagers');
        console.log('💰 Total Volume:', data.LIST.reduce((s, w) => s + (parseFloat(w.AmountWagered) || 0), 0) / 100);
        console.log('🎉 Data is now stored in KV!');
        console.log('📍 Open dashboard: http://localhost:8888/dashboard-positions.html');
        
        return data;
    } catch (error) {
        console.error('❌ Error:', error);
        console.log('💡 Try opening this page FROM fantasy402.com to include auth cookies');
    }
})();

  6. Run the modified curl command
